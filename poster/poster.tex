  \headerbox{Introduction}{name=motivation, column=0, row=0, span=3}{
    \boxtextsize
    \begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=10pt]
      \item Passive acoustic monitoring (PAM) enables large-scale biodiversity monitoring, but reliable species detection requires well-calibrated uncertainty estimates from deep learning classifiers
      \item Calibration is challenged by overlapping vocalisations, long-tailed species distributions, and distribution shifts between training and deployment data
      \item Overconfident models can bias occupancy predictions, while underconfident models reduce utility for expert review---reliable uncertainty is critical for transparent environmental decision-making
      \item The calibration of multi-label deep learning classifiers in bioacoustics has not yet been systematically assessed
    \end{itemize}
  }

  \headerbox{Benchmarking the Calibration of Bird Sound Classifiers}{name=benchmarking, span=6, below=motivation, column=0}{
    \boxtextsize
    % Three diagrams side-by-side on top
    \begin{minipage}[t]{\linewidth}
      \centering
      \includegraphics[width=0.32\linewidth]{../paper/figures/convnext_bs_reliability_diagram.pdf}\hfill
      \includegraphics[width=0.32\linewidth]{../paper/figures/ece-vs-sample-count.pdf}\hfill
      \includegraphics[width=0.32\linewidth]{../paper/figures/reliability-top-bottom.pdf}
    \end{minipage}
    \vspace{0.5em}
    % Experimental setting
    \textbf{Experimental setting.} \textbf{Models:} AudioProtoPNet, BirdMAE, ConvNeXt$_{\mathrm{BS}}$, Perch v2 (multi-label classifiers on BirdSet). \textbf{BirdSet:} Eight expert-annotated test datasets from PAM deployments. \textbf{Metrics:} ECE (expected calibration error), MCS (miscalibration score), OCS/UCS (over/underconfidence); cmAP for discrimination.
    \vspace{0.4em}
    % Results summary
    \RaggedRight
    \textbf{Results.} Globally, Perch v2 and ConvNeXt$_{\mathrm{BS}}$ are underconfident; AudioProtoPNet and BirdMAE are overconfident. Calibration varies significantly per dataset; combined ECE can mask domain-specific miscalibration, so per-dataset evaluation is essential. Class-wise ECE (centre figure) shows Perch v2 and ConvNeXt$_{\mathrm{BS}}$ with low, well-structured ECE; AudioProtoPNet and BirdMAE exhibit larger spread. For common vs.\ rare species (right), rare classes shift toward overconfidence for AudioProtoPNet and BirdMAE; Perch v2 and ConvNeXt$_{\mathrm{BS}}$ remain underconfident for both subsets. Class-wise calibration appears better for less-frequent classes.
  }

  \headerbox{Post-hoc Calibration}{name=posthoc, span=3, below=benchmarking, above=foottext, column=0}{
    \boxtextsize
    \begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=10pt]
      \item Temperature scaling (TS) and Platt scaling (PS) evaluated with two calibration sets: global parameters on POW and per-class parameters on the first 10 minutes of each test dataset
      \item Per-class Platt scaling with 10 minutes of labelled data provides the most reliable calibration improvement
      \item Global scaling benefits Perch v2 and ConvNeXt$_{\mathrm{BS}}$ but can degrade AudioProtoPNet and BirdMAE on certain datasets
      \item No method consistently outperforms across all models and datasets---calibration improvement is domain-specific
    \end{itemize}
  }

  \headerbox{Conclusions}{name=conclusions, span=3, below=benchmarking, above=foottext, column=3}{
    \boxtextsize
    \begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=10pt]
      \item Calibration varies substantially across datasets and classes; aggregating across domains can mask systematic miscalibration
      \item Perch v2 and ConvNeXt$_{\mathrm{BS}}$ consistently underconfident; AudioProtoPNet and BirdMAE show mixed over-/underconfidence
      \item Simple post hoc methods (Platt scaling with 10 min labelled data) can significantly improve deployment-specific calibration
      \item Recommend reporting OCS/UCS alongside ECE or MCS to reveal whether errors stem from over- or underconfidence
      \item Future work: Bayesian approaches, calibration@k metrics, and evaluation across additional deployment scenarios
    \end{itemize}
  }
