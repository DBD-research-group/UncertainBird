    \begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=10pt]
      \item \textbf{C1.} \textbf{Benchmark calibration} of four state-of-the-art multi-label bird sound classifiers on BirdSet~\cite{birdset}, reporting global, per-dataset and per-class calibration with threshold-free metrics
      \item \textbf{C2.} \textbf{Investigate post hoc strategies} (temperature~\cite{guo2017calibration} and Platt scaling~\cite{platt1999probabilistic}): Platt scaling with 10 minutes of test data can significantly improve calibration
      \item \textbf{C3.} \textbf{Summarise implications} to give \mbox{practitioners} concrete suggestions for deployment-specific calibration evaluation and improvement.

      \textbf{Experimental setting:}
      \begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=10pt]
        \item \textbf{Models:} AudioProtoPNet~\cite{heinrich2025audioproto}, BirdMAE~\cite{birdmae}, ConvNeXt$_{\mathrm{BS}}$~\cite{birdset}, Perch v2~\cite{perchv2} (multi-label classifiers on BirdSet~\cite{birdset}).
        \item \textbf{BirdSet:} Eight expert-annotated test datasets from PAM deployments.
        \item \textbf{Metrics:} ECE (expected calibration error), MCS (miscalibration score), OCS/UCS (over/underconfidence); cmAP for discrimination.
      \end{itemize}

    \vspace{0.4em}
    % Results summary
    \RaggedRight
    \textbf{Results:}
    \begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=10pt]
      \item Globally, Perch v2 and ConvNeXt$_{\mathrm{BS}}$ are underconfident; AudioProtoPNet and BirdMAE are overconfident (Fig.~\ref{fig:reliability-combined}).
      \item \textit{Calibration varies significantly per dataset} (Fig.~\ref{fig:convnext-reliability}); combined ECE can mask domain-specific miscalibration.
      \item Class-wise ECE (Fig.~\ref{fig:ece-vs-sample}) shows Perch v2 and ConvNeXt$_{\mathrm{BS}}$ with low ECE, increasing with the number of samples per class.
      \item For common vs.\ rare species (Fig.~\ref{fig:reliability-top-bottom}), rare classes shift toward overconfidence for AudioProtoPNet and BirdMAE; Perch v2 and ConvNeXt$_{\mathrm{BS}}$ remain underconfident for both subsets.
    \end{itemize}
  }

  \headerbox{Post Hoc Calibration}{name=posthoc, span=3, below=benchmarking, above=foottext, column=0}{
    \boxtextsize
    \begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=10pt]
      \item \textbf{Temperature scaling} (TS) and \textbf{Platt scaling} (PS) evaluated with two calibration sets: global parameters on POW and per-class parameters on the first 10 minutes of each test dataset.
      \item Per-class Platt scaling provides the most consistent calibration improvement across models and datasets, while even global scaling can improve calibration.
    \end{itemize}

  \headerbox{Conclusions}{name=conclusions, span=3, below=benchmarking, above=foottext, column=3}{
    \boxtextsize
    \begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=10pt]
      \item \textbf{Calibration varies substantially across datasets and classes;} aggregating across domains can mask miscalibration
      \item \textbf{Perch v2 and ConvNeXt$_{\mathrm{BS}}$ consistently underconfident;} AudioProtoPNet, BirdMAE show mixed over-/underconfidence
      \item \textbf{Simple post hoc methods} (Platt scaling with 10 min labelled data) can \textbf{significantly improve deployment-specific calibration}
      \item Recommend reporting OCS/UCS alongside ECE or MCS to reveal whether errors stem from over- or underconfidence
      \item \textbf{Future work:} Bayesian approaches, calibration@k metrics, and evaluation across additional deployment scenarios
    \end{itemize}
